   ================================================

## ПОДКЛЮЧЕНИЕ К БД

В PostgreSQL обычно работаем с:

database
└── tables
├── columns
└── rows

## СОЗДАНИЕ ТАБЛИЦЫ

CREATE TABLE employees (
id SERIAL PRIMARY KEY,
name VARCHAR(100) NOT NULL,
age INTEGER,
position VARCHAR(100),
salary NUMERIC(10, 2)
);

Добавить запись:

INSERT INTO employees (name, age, position, salary)
VALUES ('Михаил', 20, 'Developer', 100000);

Получить всё:

SELECT * FROM employees;

Получить конкретные столбцы:

SELECT id, name, position
FROM employees;

Условие:

SELECT *
FROM employees
WHERE id = 1;

Несколько условий:

SELECT *
FROM employees
WHERE age > 18 AND salary > 50000;

Сортировка:

SELECT *
FROM employees
ORDER BY salary DESC;

Ограничение количества:

SELECT *
FROM employees
LIMIT 10;

Поиск:

SELECT *
FROM employees
WHERE name ILIKE '%михаил%';

---

## UPDATE

UPDATE employees
SET salary = 120000
WHERE id = 1;

Важно:

UPDATE без WHERE изменит ВСЕ записи.

---

## DELETE

DELETE FROM employees
WHERE id = 1;

Важно:

DELETE без WHERE удалит ВСЕ записи.

---

## COUNT

SELECT COUNT(*)
FROM employees;

---

## NULL

SELECT *
FROM employees
WHERE position IS NULL;

НЕ:

WHERE position = NULL

Для NULL используются:

IS NULL
IS NOT NULL

---

## AND / OR

SELECT *
FROM employees
WHERE age > 18
AND salary > 50000;

SELECT *
FROM employees
WHERE position = 'Developer'
OR position = 'Designer';

---

## IN

SELECT *
FROM employees
WHERE position IN ('Developer', 'Designer');

---

## BETWEEN

SELECT *
FROM employees
WHERE salary BETWEEN 50000 AND 100000;

---

## LIKE / ILIKE

LIKE — чувствителен к регистру.

ILIKE — обычно удобнее для поиска в PostgreSQL.

SELECT *
FROM employees
WHERE name ILIKE '%иван%';

% означает любое количество символов.

'_'
означает один любой символ.

2. JOIN — САМОЕ ВАЖНОЕ
======================

Допустим:

employees

id | name | department_id

departments

id | name

Получить сотрудника и название отдела:

SELECT
employees.name,
departments.name
FROM employees
JOIN departments
ON employees.department_id = departments.id;

Чаще используют алиасы:

SELECT
e.name,
d.name
FROM employees e
JOIN departments d
ON e.department_id = d.id;

LEFT JOIN:

SELECT
e.name,
d.name
FROM employees e
LEFT JOIN departments d
ON e.department_id = d.id;

Разница:

JOIN / INNER JOIN
только записи, для которых найдено соответствие.

LEFT JOIN
все записи из левой таблицы + соответствие справа, если оно есть.

3. ПАРАМЕТРЫ SQL
================

НИКОГДА не собирай SQL так:

query = f"SELECT * FROM employees WHERE id = {id}"

Или:

query = "SELECT * FROM employees WHERE name = '" + name + "'"

Это SQL injection.

Правильно:

query = "SELECT * FROM employees WHERE id = %s"
cursor.execute(query, (id,))

Несколько параметров:

query = """
SELECT *
FROM employees
WHERE age > %s
AND salary > %s
"""

cursor.execute(query, (18, 50000))

ВАЖНО:

Для одного параметра всё равно нужна запятая:

cursor.execute(
"SELECT * FROM employees WHERE id = %s",
(employee_id,)
)

4. PSYCOPG2
===========

Установка:

pip install psycopg2-binary

Импорт:

import psycopg2

---

## ПОДКЛЮЧЕНИЕ

conn = psycopg2.connect(
host="localhost",
port=5432,
database="mydb",
user="postgres",
password="password"
)

Создаём cursor:

cursor = conn.cursor()

Выполняем SQL:

cursor.execute(
"SELECT * FROM employees"
)

Получаем результат:

rows = cursor.fetchall()

Пример:

for row in rows:
print(row)

Одна запись:

row = cursor.fetchone()

---

## INSERT

cursor.execute(
"""
INSERT INTO employees (name, age, position, salary)
VALUES (%s, %s, %s, %s)
""",
("Михаил", 20, "Developer", 100000)
)

conn.commit()

ВАЖНО:

После INSERT / UPDATE / DELETE:

conn.commit()

Иначе изменения могут не сохраниться.

---

## UPDATE

cursor.execute(
"""
UPDATE employees
SET name = %s,
age = %s,
position = %s,
salary = %s
WHERE id = %s
""",
(name, age, position, salary, employee_id)
)

conn.commit()

---

## DELETE

cursor.execute(
"DELETE FROM employees WHERE id = %s",
(employee_id,)
)

conn.commit()

---

## Закрытие

cursor.close()
conn.close()

Лучше:

with conn.cursor() as cursor:
cursor.execute(...)

17. REDIRECT
============

После POST обычно не нужно оставлять пользователя на странице обработки.

Вместо:

return "Сотрудник добавлен"

делаем:

from flask import redirect, url_for

return redirect(
url_for("employees")
)

То есть:

POST
↓
INSERT
↓
REDIRECT
↓
GET /employees

Это стандартный подход.

22. ОБРАБОТКА ОШИБОК
====================

try:

```
conn = get_db_connection()

with conn.cursor() as cursor:

    cursor.execute(...)

conn.commit()
```

except Exception:

```
conn.rollback()
```

finally:

```
conn.close()
```

Если ошибка произошла после изменения БД:

conn.rollback()

отменяет транзакцию.

app.py:

from flask import (
Flask,
render_template,
request,
redirect,
url_for
)

import psycopg2

from psycopg2.extras import (
RealDictCursor
)

app = Flask(**name**)

def get_db_connection():

```
return psycopg2.connect(
    host="localhost",
    port=5432,
    database="mydb",
    user="postgres",
    password="password"
)
```

@app.route("/employees")
def employees():

```
conn = get_db_connection()

with conn.cursor(
    cursor_factory=RealDictCursor
) as cursor:

    cursor.execute(
        """
        SELECT *
        FROM employees
        ORDER BY id
        """
    )

    employees = cursor.fetchall()

conn.close()

return render_template(
    "employees.html",
    employees=employees
)
```

@app.route(
"/employees/add",
methods=["POST"]
)
def add_employee():

```
name = request.form.get("name")
age = request.form.get("age")
position = request.form.get("position")
salary = request.form.get("salary")

conn = get_db_connection()

with conn.cursor() as cursor:

    cursor.execute(
        """
        INSERT INTO employees
        (name, age, position, salary)
        VALUES (%s, %s, %s, %s)
        """,
        (
            name,
            age,
            position,
            salary
        )
    )

conn.commit()
conn.close()

return redirect(
    url_for("employees")
)
```

@app.route(
"/employees/[int:employee_id](int:employee_id)/delete",
methods=["POST"]
)
def delete_employee(employee_id):

```
conn = get_db_connection()

with conn.cursor() as cursor:

    cursor.execute(
        """
        DELETE FROM employees
        WHERE id = %s
        """,
        (employee_id,)
    )

conn.commit()
conn.close()

return redirect(
    url_for("employees")
)
```

    debug=True
)
